function stopVideo() { const frame = document.getElementById('video-frame'); frame.src = ''; } function navigateTo(viewId, pushState = true) { if (viewId !== 'player') { stopVideo(); } document.querySelectorAll('.page-view').forEach(el => el.classList.remove('active')); document.getElementById(`view-${viewId}`).classList.add('active'); const backBtn = document.getElementById('back-btn'); backBtn.style.display = viewId === 'home' ? 'none' : 'block'; if (pushState) { navigationHistory.push(viewId); } }
